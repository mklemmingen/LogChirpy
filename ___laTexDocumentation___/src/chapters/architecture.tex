// Detailed tensor processing for BirdNET implementation
const preprocessAudioTensor = async (audioSamples: Float32Array) => {
  // 1. Resampling to exact 48kHz
  const resampledAudio = await resampleAudio(audioSamples, targetSampleRate);
  
  // 2. Windowing with Hann window
  const windowedAudio = applyHannWindow(resampledAudio);
  
  // 3. STFT (Short-Time Fourier Transform)
  const stft = computeSTFT(windowedAudio, {
    nFft: 1024,
    hopLength: 512,
    window: 'hann'
  });
  
  // 4. Mel-Filter Bank Application
  const melSpectrogram = applyMelFilterBank(stft, {
    nMels: 64,
    fMin: 150,
    fMax: 15000,
    sampleRate: 48000
  });
  
  // 5. Logarithmic scaling
  const logMelSpec = melSpectrogram.map(frame => 
    frame.map(value => Math.log(Math.max(value, 1e-10)))
  );
  
  // 6. Normalization [-1, 1]
  const normalizedSpec = normalizeSpectrogram(logMelSpec);
  
  // 7. Tensor formatting for TFLite
  const inputTensor = tf.tensor4d(
    normalizedSpec, 
    [1, 288, 64, 1], // [batch, time, frequency, channels]
    'float32'
  );
  
  return inputTensor;
};

// Final optimized TFLite integration implementation
import { TensorflowModel } from 'react-native-fast-tflite';

class BirdNETClassifier {
  private primaryModel: TensorflowModel;
  private metaModel: TensorflowModel;
  
  async initialize() {
    // Model loading with optimized parameters
    this.primaryModel = await TensorflowModel.loadFromBundle({
      modelPath: 'BirdNET_GLOBAL_6K_V2.4_Model_FP32.tflite',
      numThreads: 4, // Optimized for mobile CPUs
      useGPU: false, // CPU for consistency
      useNNAPI: true // Android Neural Networks API
    });
    
    this.metaModel = await TensorflowModel.loadFromBundle({
      modelPath: 'BirdNET_GLOBAL_6K_V2.4_MData_Model_FP16.tflite',
      numThreads: 2,
      useGPU: false,
      useNNAPI: true
    });
  }
  
  async classify(audioTensor: tf.Tensor, location?: Location) {
    // Primary model inference
    const primaryResults = await this.primaryModel.predict(audioTensor);
    
    // Meta-model enhancement (if location available)
    if (location) {
      const weekOfYear = this.calculateWeekOfYear(new Date());
      const weekCosine = Math.cos(2 * Math.PI * weekOfYear / 52);
      
      const metaInput = tf.tensor2d([[
        location.latitude / 90.0,  // Normalized [-1, 1]
        location.longitude / 180.0, // Normalized [-1, 1]
        weekCosine
      ]]);
      
      const metaResults = await this.metaModel.predict(metaInput);
      
      // Probability blending (30% meta-influence)
      const blendedResults = this.blendPredictions(
        primaryResults, 
        metaResults, 
        0.3
      );
      
      return blendedResults;
    }
    
    return primaryResults;
  }
}
\end{lstlisting}

\subsection{Wikipedia Image Extraction Pipeline}

\textbf{Comprehensive Data Extraction System for 30,000+ Avian Species:}

\begin{lstlisting}[style=javascript, caption=Wikipedia Scraping Implementation, label=lst:wiki_scraping]
// Fully automated Wikipedia scraping for all bird species
class WikipediaScrapingService {
  private readonly RATE_LIMIT = 100; // Requests per minute
  private readonly RETRY_ATTEMPTS = 3;
  private readonly USER_AGENT = 'LogChirpy/1.0 (Educational Research)';
  
  async scrapeAllBirdImages(speciesList: Species[]) {
    const results = [];
    const rateLimiter = new RateLimiter(this.RATE_LIMIT);
    
    for (const species of speciesList) {
      await rateLimiter.wait();
      
      try {
        // Multi-Language Wikipedia Scraping
        const imageData = await this.scrapeSpeciesImages(species);
        results.push(imageData);
        
        // Progress tracking
        console.log(`Scraped ${results.length}/${speciesList.length}: ${species.scientific_name}`);
        
      } catch (error) {
        console.error(`Failed to scrape ${species.scientific_name}:`, error);
        // Retry mechanism
        await this.retryWithBackoff(species, error);
      }
    }
    
    return results;
  }
  
  async scrapeSpeciesImages(species: Species) {
    // 1. Primäre Wikipedia-Suche (Englisch)
    let imageUrl = await this.searchWikipediaImage(
      species.scientific_name, 'en'
    );
    
    // 2. Fallback auf Common Name Search
    if (!imageUrl && species.common_name_en) {
      imageUrl = await this.searchWikipediaImage(
        species.common_name_en, 'en'
      );
    }
    
    // 3. Multi-Language Fallback
    if (!imageUrl) {
      const languages = ['de', 'es', 'fr'];
      for (const lang of languages) {
        const commonName = species[`common_name_${lang}`];
        if (commonName) {
          imageUrl = await this.searchWikipediaImage(commonName, lang);
          if (imageUrl) break;
        }
      }
    }
    
    // 4. Wikimedia Commons Search
    if (!imageUrl) {
      imageUrl = await this.searchWikimediaCommons(species.scientific_name);
    }
    
    // 5. Quality validation
    if (imageUrl) {
      const isValid = await this.validateImageQuality(imageUrl);
      if (!isValid) {
        imageUrl = await this.findAlternativeImage(species);
      }
    }
    
    return {
      scientific_name: species.scientific_name,
      image_url: imageUrl,
      source: 'wikipedia',
      scraped_at: new Date().toISOString(),
      quality_score: await this.calculateImageQuality(imageUrl)
    };
  }
  
  private async searchWikipediaImage(query: string, language: string) {
    const apiUrl = `https://${language}.wikipedia.org/api/rest_v1/page/summary/${encodeURIComponent(query)}`;
    
    const response = await fetch(apiUrl, {
      headers: { 'User-Agent': this.USER_AGENT }
    });
    
    if (response.ok) {
      const data = await response.json();
      return data.thumbnail?.source || data.originalimage?.source;
    }
    
    return null;
  }
  
  private async searchWikimediaCommons(scientificName: string) {
    const apiUrl = `https://commons.wikimedia.org/w/api.php`;
    const params = new URLSearchParams({
      action: 'query',
      format: 'json',
      list: 'search',
      srsearch: `${scientificName} bird`,
      srnamespace: '6', // File namespace
      srlimit: '10'
    });
    
    const response = await fetch(`${apiUrl}?${params}`);
    const data = await response.json();
    
    if (data.query?.search?.length > 0) {
      // Return highest quality image
      return await this.getCommonsImageUrl(data.query.search[0].title);
    }
    
    return null;
  }
}

\subsection{Mass Translation System for 30,000+ Species}

\textbf{Multi-API Translation Pipeline with Fallback Strategies:}

\begin{lstlisting}[style=javascript, caption=Automated Translation System, label=lst:translation_system]
// Comprehensive translation system with multiple API fallbacks
class MassTranslationService {
  private translationAPIs = [
    { name: 'DeepL', quota: 500000, cost: 'low' },
    { name: 'Google Translate', quota: 100000, cost: 'medium' },
    { name: 'Microsoft Translator', quota: 2000000, cost: 'free' },
    { name: 'LibreTranslate', quota: 'unlimited', cost: 'free' }
  ];
  
  async translateAllSpecies(species: Species[], targetLanguages: string[]) {
    const translationBatches = this.createBatches(species, 100); // Batch processing
    const results = [];
    
    for (const batch of translationBatches) {
      console.log(`Processing batch ${results.length + 1}/${translationBatches.length}`);
      
      const batchResults = await Promise.allSettled(
        batch.map(sp => this.translateSpecies(sp, targetLanguages))
      );
      
      results.push(...batchResults);
      
      // Rate limiting zwischen Batches
      await this.delay(1000);
    }
    
    return this.processBatchResults(results);
  }
  
  async translateSpecies(species: Species, languages: string[]) {
    const translations = { scientific_name: species.scientific_name };
    
    for (const lang of languages) {
      if (species[`common_name_${lang}`]) {
        // Bereits vorhanden, überspringen
        translations[`common_name_${lang}`] = species[`common_name_${lang}`];
        continue;
      }
      
      try {
        // Primäre Übersetzung von Englisch
        let translation = await this.translateWithFallback(
          species.common_name_en,
          'en',
          lang
        );
        
        // Validierung der Übersetzung
        if (!this.isValidBirdName(translation, lang)) {
          // Fallback: Wissenschaftlicher Name + Sprachspezifische Suffixe
          translation = await this.generateLocalizedName(
            species.scientific_name,
            lang
          );
        }
        
        translations[`common_name_${lang}`] = translation;
        
      } catch (error) {
        console.error(`Translation failed for ${species.scientific_name} to ${lang}:`, error);
        // Fallback: Behalte englischen Namen
        translations[`common_name_${lang}`] = species.common_name_en;
      }
    }
    
    return translations;
  }
  
  private async translateWithFallback(text: string, from: string, to: string) {
    for (const api of this.translationAPIs) {
      try {
        if (this.hasQuotaRemaining(api)) {
          const translation = await this.callTranslationAPI(api, text, from, to);
          if (translation && translation !== text) {
            this.updateQuotaUsage(api);
            return translation;
          }
        }
      } catch (error) {
        console.warn(`${api.name} failed, trying next API:`, error);
      }
    }
    
    // Letzte Fallback-Option: Regel-basierte Übersetzung
    return await this.ruleBasedTranslation(text, to);
  }
  
  private async ruleBasedTranslation(englishName: string, targetLang: string) {
    // Sprachspezifische Transformationsregeln
    const rules = {
      'de': {
        patterns: [
          { from: /(.+) Warbler$/, to: '$1-Waldsänger' },
          { from: /(.+) Flycatcher$/, to: '$1-Fliegenschnäpper' },
          { from: /(.+) Hawk$/, to: '$1-Falke' },
          { from: /(.+) Owl$/, to: '$1-Eule' }
        ],
        fallback: name => `${name} (Vogel)`
      },
      'es': {
        patterns: [
          { from: /(.+) Warbler$/, to: 'Reinita $1' },
          { from: /(.+) Flycatcher$/, to: 'Papamoscas $1' },
          { from: /(.+) Hawk$/, to: 'Halcón $1' }
        ],
        fallback: name => `${name} (ave)`
      }
      // ... weitere Sprachen
    };
    
    const langRules = rules[targetLang];
    if (!langRules) return englishName;
    
    // Pattern-Matching anwenden
    for (const rule of langRules.patterns) {
      if (rule.from.test(englishName)) {
        return englishName.replace(rule.from, rule.to);
      }
    }
    
    // Fallback-Regel
    return langRules.fallback(englishName);
  }

// Advanced string comparison algorithms for BirDex search
class SpeciesStringMatcher {
  
  async findBestMatch(query: string, language: string = 'en'): Promise<Species | null> {
    // 1. Exact match (fastest path)
    let match = await this.exactMatch(query, language);
    if (match) return match;
    
    // 2. Normalized search
    match = await this.normalizedMatch(query, language);
    if (match) return match;
    
    // 3. Subspecies-Fallback
    match = await this.subspeciesFallback(query, language);
    if (match) return match;
    
    // 4. Fuzzy-Matching mit verschiedenen Algorithmen
    match = await this.fuzzyMatch(query, language);
    if (match) return match;
    
    // 5. Cross-Language-Suche
    match = await this.crossLanguageSearch(query);
    if (match) return match;
    
    return null;
  }
  
  private async fuzzyMatch(query: string, language: string): Promise<Species | null> {
    const candidates = await this.getAllSpeciesNames(language);
    let bestMatch = null;
    let bestScore = 0;
    
    for (const candidate of candidates) {
      // Multiple Similarity-Algorithmen kombinieren
      const levenshteinScore = this.levenshteinSimilarity(query, candidate.name);
      const jaccardScore = this.jaccardSimilarity(query, candidate.name);
      const jaroWinklerScore = this.jaroWinklerSimilarity(query, candidate.name);
      const phoneticScore = this.phoneticSimilarity(query, candidate.name);
      
      // Gewichtete Kombination der Scores
      const combinedScore = (
        levenshteinScore * 0.3 +
        jaccardScore * 0.2 +
        jaroWinklerScore * 0.3 +
        phoneticScore * 0.2
      );
      
      if (combinedScore > bestScore && combinedScore > 0.7) {
        bestScore = combinedScore;
        bestMatch = candidate;
      }
    }
    
    return bestMatch ? await this.getSpeciesByName(bestMatch.name) : null;
  }
  
  private levenshteinSimilarity(a: string, b: string): number {
    const distance = this.levenshteinDistance(a.toLowerCase(), b.toLowerCase());
    const maxLength = Math.max(a.length, b.length);
    return maxLength === 0 ? 1 : (maxLength - distance) / maxLength;
  }
  
  private jaccardSimilarity(a: string, b: string): number {
    const setA = new Set(a.toLowerCase().split(''));
    const setB = new Set(b.toLowerCase().split(''));
    const intersection = new Set([...setA].filter(x => setB.has(x)));
    const union = new Set([...setA, ...setB]);
    return intersection.size / union.size;
  }
  
  private jaroWinklerSimilarity(a: string, b: string): number {
    // Implementierung des Jaro-Winkler-Algorithmus
    const jaroSim = this.jaroSimilarity(a, b);
    if (jaroSim < 0.7) return jaroSim;
    
    // Berechne gemeinsame Präfix-Länge (max 4 Zeichen)
    let prefixLength = 0;
    for (let i = 0; i < Math.min(a.length, b.length, 4); i++) {
      if (a[i].toLowerCase() === b[i].toLowerCase()) {
        prefixLength++;
      } else {
        break;
      }
    }
    
    return jaroSim + (0.1 * prefixLength * (1 - jaroSim));
  }
  
  private phoneticSimilarity(a: string, b: string): number {
    // Soundex-ähnlicher phonetischer Vergleich für Vogelnamen
    const phoneticA = this.generatePhoneticCode(a);
    const phoneticB = this.generatePhoneticCode(b);
    return phoneticA === phoneticB ? 1 : 0;
  }
  
  private async crossLanguageSearch(query: string): Promise<Species | null> {
    const languages = ['en', 'de', 'es', 'fr', 'uk', 'ar'];
    
    for (const lang of languages) {
      const match = await this.fuzzyMatch(query, lang);
      if (match) {
        console.log(`Cross-language match found in ${lang}`);
        return match;
      }
    }
    
    return null;
  }
}
